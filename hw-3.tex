    \begin{question}{Cookie Connoisseur}
        \begin{part}
            Batch sizes: $S = [1, 3, 4]$ \\
            Cookies: $n = 6$

            \begin{itemize}
                \item Greedy would give the largest batch first, making the next batches two 1's, so it would be $4 + 1 + 1 = 6$, or 3 boxes.
                \item Better: get two batches of 3, $3 + 3 = 6$, or 2 boxes.
                \item Greedy fails because it takes the max value, leaving a remainder that can only be filled by ones.
            \end{itemize}
        \end{part}
        \begin{part}
            \[
            \mathit{cook}(S[\,], i, n) = \begin{cases}
                \min\big(1 + \mathit{cook}(S[\,], i, n - S[i]),\ \mathit{cook}(S[\,], i - 1, n)\big), & S[i] \le n \land i \neq 1 \\
                \mathit{cook}(S[\,], i - 1, n), & S[i] > n \land n \neq 0 \\
                n, & n = 0 \lor i = 1
            \end{cases}
            \]
        \end{part}
        \begin{part}
            For the parameters, $S[\,]$ is the ordered list of batch sizes, $i$ is the current batch size, expecting to start at the end of $S[\,]$, and $n$ is the number of cookies requested. We expect $n$ to be greater than or equal to zero. If the current batch size is less than $n$, we return the minimum number of boxes between continuing with the current size (incrementing the total by one) and the next size down. If the current batch size is greater than $n$, we call the next size down. Finally, our base case is $n = 0$ or $i = 1$, where we return $n$ to end the recurrence.
        \end{part}
        \begin{part}
            $S = [1, 3, 4]$, $n = 6$
        \end{part}
        \begin{part}
            A $\text{length}(S) \times (n + 1)$ matrix, with the rows as the batch sizes in $S$ in increasing order.
        \end{part}
        \begin{part}
            First, we fill column 0 with 0s in every row (because 0 cookies need 0 boxes).
            Fill row 1 from 0...$n$ with the value j (because only single cookies are available).
            For each row from 2 up to length(S), we fill the columns from 1...$n$ left to right. 
            If the batch size $>$ the current column, we store the value of the row above in the same column.
            If the batch size $\leq$ the current column, we store the smaller of two options: using the batch ($1 +$ the value in the same row, $S[i]$ columns to the left) 
            or skipping it (the value in the row above in the same column).
            Each cell depends on the cell above and to its left.
            The solution is the bottom right cell.
        \end{part}
        \begin{part}
            $\Theta(nb)$
        \end{part}
    \end{question}
    \begin{question}{School Satisfaction}
        \begin{part}
            Our recurrences are maximizing the value of Miya going to class or not going to class, with $S[\,]$ being the satisfaction list. We have 4 recurrences to keep track of consecutive weeks skipped, with penalties applied accordingly. We assume $|S| > 0$.
        \end{part}
        \begin{part}
            \[
            ms_1(S[\,]) = \begin{cases}
                \max\big(S[1] + ms_1(S[2:]),\ ms_2(S[2:])\big), &  \text{length}(S) \geq 2 \\
                \max\big(S[1], 0), & \text{length}(S) = 1
            \end{cases}
            \]
            \[
            ms_2(S[\,]) = \begin{cases}
                \max\big(S[1] + ms_1(S[2:]),\ ms_3S[2:] - 20)\big), &  \text{length}(S) \geq 2 \\
                \max\big(S[1], -20), & \text{length}(S) = 1 
            \end{cases}
            \]
            \[
            ms_3(S[\,]) = \begin{cases}
                \max\big(S[1] + ms_1(S[2:]),\ ms_4(S[2:])\big), & \text{length}(S) \geq 2 \\
                \max\big(S[1], 0), & \text{length}(S) = 1
            \end{cases}
            \]
            \[
            ms_4(S[\,]) = \begin{cases}
                S[1] + ms_1(S[2:]), &  \text{length}(S) \geq 2 \\
                S[1], & \text{length}(S) = 1
            \end{cases}
            \]
        \end{part}
        \begin{part}
            Each recurrence has two cases, where $i > \text{length}(S)$ (representing where more classes need to be taken) and where $i < \text{length}(S)$ (representing when only one class remains). 
            \begin{enumerate}
                \item $ms_1$ represents the case where no classes have been skipped consecutively
                \begin{enumerate}
                    \item case $\text{length}(S) \geq 2$: We take the max between skipping and not skipping the current class.
                    \item case $\text{length}(S) = 1$: We take the max between attending the current class and $0$.
                \end{enumerate}
                \item $ms_2$ represents the case where 1 class has been skipped
                \begin{enumerate}
                    \item case $\text{length}(S) \geq 2$: We take the max between attending the current class and skipping to the again with the $-20$ penalty.
                    \item case $\text{length}(S) = 1$: We take the max between attending the current class and taking the $-20$ penalty for skipping more than twice.
                \end{enumerate}
                \item $ms_3$ represents the case where 2 classes have been skipped consecutively
                \begin{enumerate}
                    \item case $\text{length}(S) \geq 2$: We take the max between attending the current class and skipping a third time with no extra penalty.
                    \item case $\text{length}(S) = 1$: We take the max between attending the current class and $0$. Crucially, the $-20$ penalty has already been applied, so we do not apply it again.
                \end{enumerate}
                \item $ms_4$ represents the case where 3 classes have been skipped consecutively
                \begin{enumerate}
                    \item case $\text{length}(S) \geq 2$: Miya must attend the current class, so we take the current value plus the remaining optimal value.
                    \item case $\text{length}(S) = 1$: Miya must attend the current class, since the next class must be attended.
                \end{enumerate}
            \end{enumerate}
            By starting with $ms_1$, we can find the optimal satisfaction value for Miya, since every recurrence will stem from $ms_1$.
        \end{part}
        \begin{part}
            A $4 \times \text{length}(S)$ matrix. With the columns representing each element in $S[]$ and the rows representing each $ms_n$ where $n \in {1, 2, 3, 4}$
        \end{part}
        \begin{part}
            Each row of the matrix represents a different level of recurrence. First, we fill the last column with the base case from $ms_1$ to $ms_4$ respectively.
            Then, for each column from right to left, we apply to $\text{length}\geq 2$ case for the recurrence each row represents. Each evaluation happens in the previously evaluated column.
            \begin{enumerate}
                \item row 1: We take the max between the current column label plus the value in row $1$ versus the value in row $2$
                \item row 2: We take the max between the current column label plus the value in row $1$ versus the value in row $3 - 20$
                \item row 3: We take the max between the current column label plus the value in row $1$ versus the value in row $4$
                \item row 4: We take the value of the current column label plus the value in row $1$
            \end{enumerate}
            The runtime takes $O(n)$ because the number of rows is constant and each evaluation happens $n$ times.
        \end{part}
        \begin{part}
            We can then find the answer at the top left of the memoization structure.
        \end{part}
    \end{question}
